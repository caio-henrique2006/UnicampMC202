## Lista encadeadas:

Exemplo

```c
#include <stdlib.h>
#include <stdio.h>
#include <errno.h>
#include <string.h>


struct no {
  int idade;
  char nome[50];
  long cpf;
  struct no *prox;
};

typedef struct no no;

no pega(no* L1, long i) {

  for (long j=0; j<i; j++)
    L1 = L1->prox;

  return *L1;
}

/*
  Insere um nó na lista que vai ocupar a posição i,
  convencionando que o cabeça esta na posição 0.
*/
int inserir(no** L1, long i, char* nome, int idade, long cpf) {

  no* novo = malloc(sizeof(no));
  if (novo == NULL)
    return 0;

  strcpy(novo->nome,nome);
  novo->idade = idade;
  novo->cpf = cpf;

  if (i == 0) {
    novo->prox = *L1;
    *L1 = novo;
  }
  else {
    no* p = *L1;
    long j = 0;
    while (j < i-1) {
      p = p->prox;
      j++;
    }

    novo->prox = p->prox;
    p->prox = novo;
  }

  return 1;
}

int inserir_no_fim(no** L1, char* nome, int idade, long cpf) {

  no* novo = malloc(sizeof(no));
  if (novo == NULL)
    return 0;

  strcpy(novo->nome,nome);
  novo->idade = idade;
  novo->cpf = cpf;

  if (*L1 == NULL) {
    novo->prox = NULL;
    *L1 = novo;
  }
  else {
    no* p = *L1;
    while (p->prox != NULL)
      p = p->prox;

    novo->prox = NULL;
    p->prox = novo;
  }

  return 1;
}

int inserir_no_inicio(no** L1, char* nome, int idade, long cpf) {

  no* novo = malloc(sizeof(no));
  if (novo == NULL)
    return 0;

  strcpy(novo->nome,nome);
  novo->idade = idade;
  novo->cpf = cpf;

  novo->prox = *L1;
  *L1 = novo;

  return 1;
}

void imprimir(no* L1) {

  while (L1 != NULL) {
    printf("%s %d %ld\n",L1->nome,L1->idade,L1->cpf);
    L1 = L1->prox;
  }

  for ( ; L1 != NULL; L1 = L1->prox)
    printf("%s %d %ld\n",L1->nome,L1->idade,L1->cpf);

  for (no* p = L1 ; p != NULL; p = p->prox)
    printf("%s %d %ld\n",p->nome,p->idade,p->cpf);

}

void limpa(no* L1) {

  while (L1 != NULL) {
    no* p = L1;
    L1 = L1->prox;
    free(p);
  }
}

int main(void) {

  no* L = NULL;

  inserir(&L,0,"Pessoa 1", 11, 123457890);
  inserir(&L,1,"Pessoa 2", 21, 213457890);
  inserir(&L,2,"Pessoa 3", 31, 321457890);
  inserir(&L,3,"Pessoa 4", 41, 432157890);
  inserir(&L,2,"Pessoa 5", 51, 543217890);

  imprimir(L);

  no x = pega(L,2);

  printf("-->%s %d %ld\n",x.nome,x.idade,x.cpf);

  limpa(L);
  L = NULL;
}
```
